# Calculate Pressure and SOF score for each region for each year
pressure_df <- df_regions |>
  group_by(region_id, year) |>
  summarise(
    disturbed_sum   = sum(disturbed_area, na.rm = TRUE),
    soft_bottom_sum = sum(soft_bottom_area, na.rm = TRUE),
    Pressure        = disturbed_sum / soft_bottom_sum,
    SOF_score       = 1 - Pressure,
    Pressure        = Pressure * 100,
    SOF_score       = SOF_score * 100,  
    .groups = "drop"
  )

# Trend Calculation: lm for most recent 5 years of data
year_end <- max(condition_df$year, na.rm = TRUE)
year_start <- year_end - 4
calc_trends <- function(d) {
  ys <- min(d$year)
  ye <- max(d$year)
  cs <- d$condition[d$year == ys][1]
  fit_lin <- lm(condition ~ year, data = d)
  fit_log <- lm(log(condition) ~ year, data = d)
  slope_lin <- unname(coef(fit_lin)["year"])
  slope_log <- unname(coef(fit_log)["year"])
  tibble(
    year = ye,
    year_start = ys,
    year_end = ye,
    condition_start = round(cs, 1),
    trend_pct_linear = (slope_lin / cs) * 100,              # slope / start * 100
    trend_pct_cagr   = (exp(slope_log) - 1) * 100           # slope from log model
  )
}

Soft bottom by depth class

The files above are the combined habitat. This section does not replace them, and it does not change the disturbed-proportion raster saved for rhodoliths.

The same swept-area score is calculated twice. Soft bottom of 100 m or less uses mask_depth_le_100m_1km.tif. Soft bottom deeper than 100 m uses mask_depth_gt_100m_1km.tif. Those masks already partition the marine grid: a cell with any bathymetry sample of 100 m or less is shallow, and a coastal marine cell with no in-water sample is shallow as well. Every other marine cell is deep.

Within a class, the cell calculation is unchanged. Disturbance in km² is fishing hours times 3.7 km/h times a 0.030 km headrope. The cell ratio of disturbance to soft-bottom area is capped at 1. The regional pressure is the sum of that capped disturbed area divided by the sum of soft-bottom area. Condition is 100 times 1 minus that pressure. Trend is the 2021–2025 change in condition, linear and as a compound rate.

A cell’s hours stay with that cell. Regions 1 and 5 have no cells deeper than 100 m, so they are left out of the deep files. Because the two classes do not overlap, adding their disturbed area and their soft-bottom area should recover the combined score above. Saved condition is rounded to 1 decimal. Saved pressure, on the 0–1 scale, is rounded to 2 decimals. The area columns stay unrounded.

# Regional score for one depth class.
# Rows are written only where that class has soft-bottom area.
score_depth_class <- function(mask) {
  soft_class <- soft_bottom_fraction * mask
  disturbance_class <- disturbance_raster_km2 * mask

  prop_class <- disturbance_class / soft_class
  prop_class[prop_class > 1] <- 1
  prop_class[is.infinite(prop_class)] <- 0
  disturbed_class <- prop_class * soft_class

  n_cells <- ncell(template)
  n_years <- nlyr(disturbed_class)
  class_df <- tibble(
    region_id = rep(as.integer(values(template)[, 1]), times = n_years),
    year = rep(years, each = n_cells),
    in_class = rep(values(mask)[, 1], times = n_years),
    disturbed_area = as.vector(values(disturbed_class)),
    soft_bottom_area = rep(as.vector(values(soft_class)), times = n_years)
  ) |>
    filter(in_class == 1, !is.na(region_id)) |>
    filter(!is.nan(disturbed_area), !is.nan(soft_bottom_area))

  class_df |>
    group_by(region_id, year) |>
    summarise(
      disturbed_sum = sum(disturbed_area, na.rm = TRUE),
      soft_bottom_sum = sum(soft_bottom_area, na.rm = TRUE),
      .groups = "drop"
    ) |>
    filter(soft_bottom_sum > 0) |>
    mutate(
      pressure = round(disturbed_sum / soft_bottom_sum, 2),
      condition = pmin(round((1 - disturbed_sum / soft_bottom_sum) * 100, 1), 100)
    )
}

# Same trend definition as the combined habitat.
# The log fit needs a positive condition in every year of the window.
calc_trends_depth <- function(d) {
  ys <- min(d$year)
  ye <- max(d$year)
  cs <- d$condition[d$year == ys][1]
  fit_lin <- lm(condition ~ year, data = d)
  slope_lin <- unname(coef(fit_lin)["year"])
  trend_log <- NA_real_
  if (all(d$condition > 0)) {
    fit_log <- lm(log(condition) ~ year, data = d)
    slope_log <- unname(coef(fit_log)["year"])
    trend_log <- (exp(slope_log) - 1) * 100
  }
  tibble(
    year = ye,
    year_start = ys,
    year_end = ye,
    condition_start = round(cs, 1),
    trend_pct_linear = (slope_lin / cs) * 100,
    trend_pct_cagr = trend_log
  )
}

write_depth_outputs <- function(scores, file_suffix) {
  out_dir <- here("_habitats", "HAB_soft_bottom", "SOF_outputs")
  write_csv(
    scores |>
      transmute(
        region_id,
        year,
        disturbed_sum,
        soft_bottom_sum,
        Pressure = round(100 - condition, 1),
        SOF_score = condition
      ),
    file.path(out_dir, paste0("sof_region_scores_per_year_", file_suffix, ".csv"))
  )
  write_csv(
    scores |> select(region_id, year, condition),
    file.path(out_dir, paste0("sof_status_", file_suffix, ".csv"))
  )
  write_csv(
    scores |> select(region_id, year, pressure),
    file.path(out_dir, paste0("sof_pressure_", file_suffix, ".csv"))
  )

  year_end <- max(scores$year)
  trend <- scores |>
    filter(year >= year_end - 4, year <= year_end) |>
    group_by(region_id) |>
    group_modify(~ calc_trends_depth(.x)) |>
    ungroup()
  write_csv(trend, file.path(out_dir, paste0("sof_trend_", file_suffix, ".csv")))
  trend
}
